  <script>
    let currentPrice = 250;
    // ข้อมูลตัวอย่างเริ่มต้น เพื่อให้กดดูใบหยิบของและใบจ่าหน้าได้ทันที
    let orders = JSON.parse(localStorage.getItem('shampoo_orders')) || [
      {
        OrderID: "ORD-100001",
        Timestamp: new Date().toLocaleString("th-TH"),
        CustomerName: "สมหญิง รักสวย (ตัวอย่าง)",
        Phone: "0891234567",
        Address: "99/1 ซ.พหลโยธิน 34 แขวงเสนานิคม เขตจตุจักร กรุงเทพฯ 10900",
        SKU: "SHP-001",
        ProductName: "แชมพูสูตรมะกรูดและขิง",
        Qty: 2,
        TotalPrice: 500
      }
    ];

    function switchTab(tab) {
      document.getElementById('tab-shop').classList.toggle('hidden', tab !== 'shop');
      document.getElementById('tab-warehouse').classList.toggle('hidden', tab !== 'warehouse');
      if (tab === 'warehouse') renderOrders();
    }

    function selectProduct(sku, name, price) {
      document.getElementById('sku').value = sku;
      document.getElementById('productName').value = name;
      currentPrice = price;
      document.getElementById('order-form-section').scrollIntoView({ behavior: 'smooth' });
    }

    document.getElementById('orderForm').addEventListener('submit', (e) => {
      e.preventDefault();
      const qty = parseInt(document.getElementById('qty').value);
      const newOrder = {
        OrderID: "ORD-" + Math.floor(100000 + Math.random() * 900000),
        Timestamp: new Date().toLocaleString("th-TH"),
        CustomerName: document.getElementById('customerName').value,
        Phone: document.getElementById('phone').value,
        Address: document.getElementById('address').value,
        SKU: document.getElementById('sku').value,
        ProductName: document.getElementById('productName').value,
        Qty: qty,
        TotalPrice: qty * currentPrice
      };

      orders.unshift(newOrder);
      localStorage.setItem('shampoo_orders', JSON.stringify(orders));
      alert("✅ สั่งซื้อสำเร็จ! เลขที่ออเดอร์: " + newOrder.OrderID);
      document.getElementById('orderForm').reset();
      selectProduct('SHP-001', 'แชมพูสูตรมะกรูดและขิง', 250);
    });

    function renderOrders() {
      const tbody = document.getElementById('ordersTableBody');
      tbody.innerHTML = orders.map((order, index) => `
        <tr class="hover:bg-gray-50">
          <td class="p-3 font-mono font-bold text-emerald-700">${order.OrderID}</td>
          <td class="p-3">${order.CustomerName}<br><span class="text-xs text-gray-400">${order.Phone}</span></td>
          <td class="p-3"><span class="font-bold">${order.SKU}</span> - ${order.ProductName}</td>
          <td class="p-3 font-bold">${order.Qty}</td>
          <td class="p-3">฿${order.TotalPrice}</td>
          <td class="p-3 text-center">
            <button onclick="preparePack(${index})" class="bg-amber-500 hover:bg-amber-600 text-white px-3 py-1.5 rounded text-xs font-semibold">
              📦 หยิบ & จ่าหน้าพัสดุ
            </button>
          </td>
        </tr>
      `).join('');
    }

    function preparePack(index) {
      const order = orders[index];
      document.getElementById('pack-orderId').innerText = order.OrderID;
      document.getElementById('pack-sku').innerText = order.SKU;
      document.getElementById('pack-product').innerText = order.ProductName;
      document.getElementById('pack-qty').innerText = order.Qty;
      document.getElementById('pack-name').innerText = order.CustomerName;
      document.getElementById('pack-phone').innerText = "โทร: " + order.Phone;
      document.getElementById('pack-address').innerText = order.Address;
      document.getElementById('pack-ref').innerText = order.OrderID;

      const area = document.getElementById('packingArea');
      area.classList.remove('hidden');
      area.scrollIntoView({ behavior: 'smooth' });
    }

    function exportCSV() {
      const headers = ["OrderID,Timestamp,CustomerName,Phone,Address,SKU,ProductName,Qty,TotalPrice"];
      const rows = orders.map(o => `"${o.OrderID}","${o.Timestamp}","${o.CustomerName}","${o.Phone}","${o.Address}","${o.SKU}","${o.ProductName}",${o.Qty},${o.TotalPrice}`);
      const csvContent = "data:text/csv;charset=utf-8,\uFEFF" + [headers, ...rows].join("\n");
      const link = document.createElement("a");
      link.setAttribute("href", encodeURI(csvContent));
      link.setAttribute("download", "shampoo_orders.csv");
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }
